#include <tunables/global>

profile emporia-vue-ws flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  capability net_bind_service,
  capability chown,
  capability dac_override,
  capability setgid,
  capability setuid,

  # S6-Overlay
  /init rx,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** rwix,
  /package/** ix,
  /command/** ix,
  /etc/s6/** rix,
  /etc/s6-overlay/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/services.d/** rwix,
  /dev/tty rw,
  /var/run/** rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Access to options.json and other files within your addon
  /data/** rw,

  # Start main binary
  /usr/bin/ws-server cx -> ws_server,

  profile ws_server flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Network access
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,

    # Capabilities for binding to port
    capability net_bind_service,

    # Binary execution
    /usr/bin/ws-server mr,

    # Config and data access
    /data/** rw,

    # Temporary files
    /tmp/** rwk,

    # System info
    /proc/sys/net/** r,
    /etc/resolv.conf r,
    /etc/hosts r,
    /etc/nsswitch.conf r,
    /etc/ssl/** r,
    /usr/share/ca-certificates/** r,
    /etc/ca-certificates/** r,
  }
}
