#include <tunables/global>

profile ups_monitor flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  capability net_bind_service,
  capability net_raw,
  capability chown,
  capability dac_override,
  capability setuid,
  capability setgid,

  # Network access
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,

  # Data directory access
  /data/** rw,

  # Config and runtime files
  /config/** r,
  /run/** rw,
  /tmp/** rw,

  # s6-overlay init system
  /init rix,
  /command/** rix,
  /package/** rix,

  # Shell access for s6 scripts
  /bin/sh rix,
  /bin/bash rix,
  /usr/bin/bashio rix,
  /usr/bin/env rix,

  # Binary execution
  /usr/bin/ups-server rmix,
  /usr/local/bin/ups-server rmix,

  # System libraries and files
  /lib/** rm,
  /usr/lib/** rm,
  /etc/passwd r,
  /etc/group r,
  /etc/hosts r,
  /etc/resolv.conf r,
  /etc/ssl/** r,
  /etc/cont-init.d/** rix,
  /etc/services.d/** rix,

  # Proc filesystem (limited)
  /proc/sys/net/** r,

  # Deny write access to sensitive areas
  deny /etc/** w,
  deny /bin/** w,
  deny /sbin/** w,
  deny /usr/** w,
}
